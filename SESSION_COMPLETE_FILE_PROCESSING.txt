"""
NEW METHODS IN FILE_HANDLER.PY
═════════════════════════════════

1. process_audio(file_path: str) → Optional[str]
   ─────────────────────────────────────────────
   
   Purpose: Transcribe audio to text using Whisper API
   
   How it works:
   - Opens audio file (mp3, wav, ogg, m4a, flac, aac)
   - Sends to Whisper API: POST /v1/audio/transcriptions
   - Returns full transcription as string
   
   Example:
     text = handler.process_audio("message.mp3")
     # Returns: "Bonjour, j'ai une question"
   
   Cost: ~$0.006 per minute of audio
   Speed: ~1 second per minute of audio
   Accuracy: 99%+ (Whisper is state-of-the-art)
   Languages: Supports 99 languages (auto-detects)


2. extract_video_frames(file_path: str, num_frames: int = 3) → Optional[List[Dict]]
   ────────────────────────────────────────────────────────────────────────────────
   
   Purpose: Extract frames from video for Vision API analysis
   
   How it works:
   - Opens video file (mp4, avi, mov, mkv, webm, flv)
   - Uses OpenCV if available, ffmpeg as fallback
   - Extracts 3 frames at 0%, 50%, 100% positions
   - Encodes each frame as base64 image
   - Formats for Vision API consumption
   - Returns list of images ready for analysis
   
   Example:
     frames = handler.extract_video_frames("presentation.mp4", num_frames=3)
     # Returns: [frame1_image, frame2_image, frame3_image]
     # Each formatted: {'type': 'image_url', 'image_url': {...}}
   
   Video Support: mp4, avi, mov, mkv, webm, flv
   Frame Count: Configurable (default 3)
   Quality: Full resolution (no compression)


3. _extract_video_audio(file_path: str) → Optional[str]
   ──────────────────────────────────────────────────────
   
   Purpose: Extract audio track from video and transcribe
   
   How it works:
   - Opens video file
   - Uses ffmpeg to extract audio: ffmpeg -i video.mp4 -q:a 9 audio.wav
   - Calls process_audio() on extracted track
   - Returns transcribed text
   - Automatically cleans up temp files
   
   Example:
     audio_text = handler._extract_video_audio("tutorial.mp4")
     # Returns: "Step 1: Open the file..."
   
   Fallback: Used when ffmpeg is needed (always available on Windows)
   Temp Files: Automatically cleaned up


4. process_video(file_path: str) → Optional[Dict]
   ──────────────────────────────────────────────
   
   Purpose: Complete video processing (frames + audio)
   
   How it works:
   - Calls extract_video_frames()
   - Calls _extract_video_audio()
   - Combines results
   - Returns structured dict
   
   Example:
     video_data = handler.process_video("demo.mp4")
     # Returns: {
     #     'frames': [image1, image2, image3],
     #     'audio_text': 'Video transcript...',
     #     'file_name': 'demo.mp4'
     # }
   
   Returns Both:
   - frames: For Vision API visual analysis
   - audio_text: For agent text understanding


5. prepare_files_for_api() → Dict
   ──────────────────────────────
   
   Purpose: Main entry point - process ALL selected files
   
   How it works:
   - Iterates through self.selected_files
   - For each image: encode_image_to_base64()
   - For each audio: process_audio()
   - For each video: process_video()
   - Combines all results
   - Returns ready-to-use dict
   
   Returns:
   {
       'images': [...],                # Vision API images (from images + video frames)
       'audio_texts': [...],           # Transcribed audio
       'video_descriptions': [...],    # Video metadata + transcriptions
       'text_content': [...],          # Human-readable full content
       'files_info': [...]             # Metadata
   }
   
   Example:
     handler.add_file("photo.jpg")
     handler.add_file("audio.mp3")
     handler.add_file("video.mp4")
     
     files_data = handler.prepare_files_for_api()
     # Returns ALL processed content ready for agent
"""
